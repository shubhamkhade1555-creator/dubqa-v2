DubQA v2 — Complete Architecture & Codebase Handbook

System Architecture & End-to-End Pipeline

DubQA v2 is an automated audio-visual Quality Assurance system for evaluating dubbed videos against ground-truth source videos.

End-to-End Data Flow

  1. Input Ingestion: Source video & 1+ Dubbed video files uploaded via Web UI or REST API.
  2. Audio Processing: FFmpeg demuxes 16kHz mono WAV audio tracks; Energy VAD detects speech intervals.
  3. Gemini ASR Engine: Direct audio upload to Google Gemini 2.5 Flash API with structured JSON output schema for word timestamps and language ID.
  4. MediaPipe Vision Engine: Tasks Vision FaceLandmarker measures upper and lower inner lip distance normalized by mouth width over sampled video frames.
  5. Drift & Alignment Math: Dynamic Time Warping aligns words; onset/tail drifts and speech open ratios during silence are measured.
  6. Gemini LLM Judge (Optional): Evaluates emotion preservation and tone match per dialogue segment.
  7. Scoring & Pairwise Matrix: Weighted scores, Good/Bad gates, and complete head-to-head tournament rankings are computed.
  8. Artifact Generation: Local JSON matrices and a standalone interactive HTML report with synchronized video players are generated.

core/asr.py — Gemini Audio Speech Recognition

Direct audio upload to Gemini 2.5 Flash API with structured JSON output schema extracting word timestamps (start, end) and detected language. Replaces Whisper.
"""Word-level transcription and language detection with Gemini API.

Replaces faster-whisper with Google Gemini audio understanding.
Falls back cleanly to precomputed JSON files or reports errors if GEMINI_API_KEY is unset.
"""
import json
import os
import time
from pathlib import Path
from pydantic import BaseModel, Field


class WordTimestamp(BaseModel):
    word: str
    start: float = Field(description="Start time in seconds with millisecond precision, e.g. 0.45")
    end: float = Field(description="End time in seconds with millisecond precision, e.g. 0.82")


class SentenceSegment(BaseModel):
    text: str
    start: float
    end: float


class AudioTranscriptionSchema(BaseModel):
    language: str = Field(description="Detected ISO 639-1 language code, e.g. 'en', 'es', 'hi'")
    language_prob: float = Field(description="Confidence probability between 0.0 and 1.0", default=1.0)
    words: list[WordTimestamp]
    sentences: list[SentenceSegment]


def transcribe_words(wav: Path, lang: str | None = None) -> dict:
    """Transcribes audio using Gemini and extracts word-level timestamps and detected language."""
    api_key = os.environ.get("GEMINI_API_KEY")
    if not api_key:
        raise RuntimeError("GEMINI_API_KEY is not set in environment or .env file.")

    try:
        from google import genai
        from google.genai import types
    except ImportError as e:
        raise RuntimeError("google-genai library is not installed. Run: pip install google-genai") from e

    client = genai.Client(api_key=api_key)
    model_name = os.environ.get("GEMINI_MODEL", "gemini-2.5-flash")

    # 1. Upload audio to Gemini File API
    audio_file = client.files.upload(file=str(wav))
    try:
        # Wait for file to become active
        for _ in range(30):
            status = client.files.get(name=audio_file.name).state.name
            if status == "ACTIVE":
                break
            if status == "FAILED":
                raise RuntimeError(f"Audio file processing failed on Gemini server: {audio_file.name}")
            time.sleep(1)

        # 2. Build structured transcription prompt
        prompt = (
            "You are an expert audio transcription system. Listen to the attached audio carefully.\n"
            "Generate an exact, highly accurate verbatim transcript with precise word-level start and end timestamps (in seconds).\n"
            "Ensure every spoken word has its corresponding start and end offset.\n"
        )
        if lang:
            prompt += f"Expected spoken language: {lang}.\n"

        response = client.models.generate_content(
            model=model_name,
            contents=[audio_file, prompt],
            config=types.GenerateContentConfig(
                response_mime_type="application/json",
                response_schema=AudioTranscriptionSchema,
                temperature=0.0,
            ),
        )

        data = json.loads(response.text)

        # 3. Format dictionary to conform with DubQA pipeline requirements
        words = [
            {"word": w["word"].strip(), "start": round(float(w["start"]), 3), "end": round(float(w["end"]), 3)}
            for w in data.get("words", [])
        ]
        sentences = [
            {"text": s["text"].strip(), "start": round(float(s["start"]), 3), "end": round(float(s["end"]), 3)}
            for s in data.get("sentences", [])
        ]

        return {
            "words": words,
            "sentences": sentences,
            "language": data.get("language", lang or "en"),
            "language_prob": round(float(data.get("language_prob", 1.0)), 3),
        }

    finally:
        # Clean up uploaded audio file from Gemini remote storage
        try:
            client.files.delete(name=audio_file.name)
        except Exception:
            pass


def load_words(path: Path) -> dict:
    """Loads precomputed transcription JSON file (used for QC mode)."""
    return json.loads(Path(path).read_text(encoding="utf-8"))

core/lips.py — MediaPipe Lip Movement Tracking

Uses Google MediaPipe Tasks Vision FaceLandmarker on sampled video frames to calculate vertical inner lip opening normalized by mouth corner width.
"""Lip tracking.

Two sources:
  1. MediaPipe FaceLandmarker (tasks.vision API) on real video.
  2. Precomputed JSON (used by the QC test set and for re-running scoring).

Lip opening = distance(upper inner lip, lower inner lip) / distance(mouth corners).
"""
import json
import urllib.request
from pathlib import Path

import numpy as np

UPPER, LOWER, LEFT, RIGHT = 13, 14, 61, 291
MODEL_URL = "https://storage.googleapis.com/mediapipe-models/face_landmarker/face_landmarker/float16/1/face_landmarker.task"


def opening_from_landmarks(lm) -> float:
    def d(a, b):
        return float(np.hypot(lm[a][0] - lm[b][0], lm[a][1] - lm[b][1]))
    return d(UPPER, LOWER) / max(d(LEFT, RIGHT), 1e-6)


def _ensure_task_model() -> Path:
    p = Path(__file__).resolve().parent.parent / "config" / "models" / "face_landmarker.task"
    p.parent.mkdir(parents=True, exist_ok=True)
    if not p.exists() or p.stat().st_size < 1000:
        urllib.request.urlretrieve(MODEL_URL, str(p))
    return p


def track_video_mediapipe(video: Path, fps_sample: float = 25.0):
    """Return (t_seconds, opening) arrays using MediaPipe FaceLandmarker."""
    try:
        import cv2
        import mediapipe as mp
    except ImportError as e:
        raise RuntimeError("MediaPipe/OpenCV not installed; use requirements-ml.txt") from e

    cap = cv2.VideoCapture(str(video))
    src_fps = cap.get(cv2.CAP_PROP_FPS) or 25.0
    step = max(1, int(round(src_fps / fps_sample)))

    # Support both legacy mediapipe solutions and modern mediapipe.tasks.vision
    if hasattr(mp, "solutions") and hasattr(mp.solutions, "face_mesh"):
        face = mp.solutions.face_mesh.FaceMesh(static_image_mode=False, max_num_faces=1, refine_landmarks=True)
        ts, vals, idx = [], [], 0
        while True:
            ok, frame = cap.read()
            if not ok:
                break
            if idx % step == 0:
                res = face.process(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
                t = idx / src_fps
                if res.multi_face_landmarks:
                    lm = [(p.x, p.y) for p in res.multi_face_landmarks[0].landmark]
                    vals.append(opening_from_landmarks(lm))
                else:
                    vals.append(np.nan)
                ts.append(t)
            idx += 1
        cap.release()
        face.close()
        return np.array(ts), np.array(vals, dtype=float)

    # Modern mediapipe Tasks Vision API
    from mediapipe.tasks import python
    from mediapipe.tasks.python import vision

    model_path = _ensure_task_model()
    base_options = python.BaseOptions(model_asset_path=str(model_path))
    options = vision.FaceLandmarkerOptions(
        base_options=base_options,
        running_mode=vision.RunningMode.VIDEO,
        num_faces=1
    )

    ts, vals, idx = [], [], 0
    with vision.FaceLandmarker.create_from_options(options) as landmarker:
        while True:
            ok, frame = cap.read()
            if not ok:
                break
            if idx % step == 0:
                rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                mp_img = mp.Image(image_format=mp.ImageFormat.SRGB, data=rgb)
                timestamp_ms = int(round((idx / src_fps) * 1000))
                res = landmarker.detect_for_video(mp_img, timestamp_ms)
                t = idx / src_fps
                if res.face_landmarks:
                    lm = [(p.x, p.y) for p in res.face_landmarks[0]]
                    vals.append(opening_from_landmarks(lm))
                else:
                    vals.append(np.nan)
                ts.append(t)
            idx += 1
    cap.release()
    return np.array(ts), np.array(vals, dtype=float)


def load_precomputed(path: Path):
    """JSON: {"fps": 25, "opening": [...]} or {"t": [...], "opening": [...]}"""
    data = json.loads(Path(path).read_text(encoding="utf-8"))
    op = np.array(data["opening"], dtype=float)
    if "t" in data:
        t = np.array(data["t"], dtype=float)
    else:
        t = np.arange(op.size) / float(data.get("fps", 25.0))
    return t, op

core/pipeline.py — Master Orchestration Pipeline

Coordinates audio extraction, Gemini ASR, MediaPipe lip tracking, drift math, lexical alignment, LLM emotion evaluation, and saves all JSON artifacts locally.
"""End-to-end run for one batch.

mode="precomputed": uses word and lip JSON files under <run>/precomputed/ (QC and offline).
mode="full":        uses faster-whisper and MediaPipe on the real media (needs requirements-ml.txt).
"""
import json
import logging
import os
from datetime import datetime, timezone
from pathlib import Path

import numpy as np

from core import align, asr, judge_gemini, lang, lips, media, pairwise, scoring, timing, vad
from core import info as ginfo
from core.manifest import load_manifest

log = logging.getLogger("dubqa")


def _write(path: Path, data) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")


def _status(run_dir: Path, status: str, **extra) -> None:
    meta_p = run_dir / "run_meta.json"
    meta = json.loads(meta_p.read_text(encoding="utf-8")) if meta_p.exists() else {}
    meta.update({"status": status, "updated": datetime.now(timezone.utc).isoformat()}, **extra)
    _write(meta_p, meta)


def _words_and_lips(mode, run_dir, wav, video, pre_words, pre_lips, lang):
    if mode == "precomputed":
        words = asr.load_words(pre_words)
        t, op = lips.load_precomputed(pre_lips)
    else:
        words = asr.transcribe_words(wav, lang)
        t, op = lips.track_video_mediapipe(video)
    return words, t, op


def run(run_dir: Path, mode: str = "precomputed", use_judge: bool = False, use_gemini: bool = True) -> dict:
    run_dir = Path(run_dir)
    man = load_manifest(run_dir / "manifest.json")
    th = scoring.load_thresholds()
    inputs, work, pre = run_dir / "inputs", run_dir / "work", run_dir / "precomputed"
    _status(run_dir, "running", mode=mode)
    log.info("run %s started (mode=%s, judge=%s, gemini=%s)", man["run_id"], mode, use_judge, use_gemini)
    try:
        log.info("source: %s", man["source"]["file"])
        src_path = inputs / man["source"]["file"]
        src_info = media.probe(src_path)
        src_wav = media.extract_audio_wav(src_path, work / "source.wav")
        src_words, t_s, op_s = _words_and_lips(
            mode, run_dir, src_wav, src_path,
            pre / "source_words.json", pre / "source_lips.json", man["source"].get("lang"))
        # Persist extracted words and lips for full auditability
        _write(run_dir / "words" / "source_words.json", src_words)
        _write(run_dir / "lips" / "source_lips.json", {"t": t_s.tolist() if hasattr(t_s, "tolist") else list(t_s),
                                                       "opening": op_s.tolist() if hasattr(op_s, "tolist") else list(op_s)})
        src_w = src_words["words"]
        src_sent = src_words.get("sentences", [])
        src_speech_s = _speech_total(src_wav)

        src_det = lang.detect_for_words(src_w, src_words.get("language"))
        src_ext = ginfo.extract_video(src_path, run_dir / "gemini" / "source.json", use_gemini)
        if src_det["lang"] is None and src_ext.get("language"):
            src_det = {"lang": src_ext["language"], "method": "gemini", "script": None, "confidence": 1.0, "note": ""}
        src_lang, src_lang_notes = lang.resolve(man["source"].get("lang"), src_det["lang"])
        log.info("source language: %s (declared=%s, detected=%s via %s)", src_lang,
                 man["source"].get("lang"), src_det["lang"], src_det["method"])

        results = {"run_id": man["run_id"], "mode": mode, "generated": datetime.now(timezone.utc).isoformat(),
                   "thresholds_version": th["version"], "source": {**src_info, "file": src_path.name,
                   "words": len(src_w), "speech_s": src_speech_s, "language": src_lang},
                   "outputs": {}, "segments": [], "pairs": {}, "notes": [],
                   "languages": {"source": {"declared": man["source"].get("lang"), "resolved": src_lang,
                                            "detection": src_det, "notes": src_lang_notes}},
                   "gemini_source": _gemini_brief(src_ext), "info": {},
                   "required_data": ginfo.checklist(run_dir, man, mode, use_gemini)}
        # segments = source sentences (shared by every output)
        for i, s in enumerate(src_sent, start=1):
            results["segments"].append({"seg": i, "start": s["start"], "end": s["end"], "text": s["text"]})

        values_by_out: dict[str, dict] = {}
        for o in man["outputs"]:
            oid = o["output_id"]
            path = inputs / o["file"]
            log.info("measuring %s (%s)", oid, o["vendor"])
            info = media.probe(path)
            wav = media.extract_audio_wav(path, work / f"{oid}.wav")
            samples, sr = media.read_wav_mono(wav)
            speech = vad.speech_intervals(samples, sr)
            dwords, t, op = _words_and_lips(mode, run_dir, wav, path, pre / f"{oid}_words.json",
                                            pre / f"{oid}_lips.json", man.get("target_lang"))
            # Persist extracted words and lips locally for each dub output
            _write(run_dir / "words" / f"{oid}_words.json", dwords)
            _write(run_dir / "lips" / f"{oid}_lips.json", {"t": t.tolist() if hasattr(t, "tolist") else list(t),
                                                           "opening": op.tolist() if hasattr(op, "tolist") else list(op)})
            d_w = dwords["words"]
            dub_det = lang.detect_for_words(d_w, dwords.get("language"))
            dub_ext = ginfo.extract_video(path, run_dir / "gemini" / f"{oid}.json", use_gemini)
            if dub_det["lang"] is None and dub_ext.get("language"):
                dub_det = {"lang": dub_ext["language"], "method": "gemini", "script": None, "confidence": 1.0, "note": ""}
            dub_lang, dub_lang_notes = lang.resolve(man.get("target_lang"), dub_det["lang"])
            log.info("%s language: %s (declared=%s, detected=%s via %s)", oid, dub_lang,
                     man.get("target_lang"), dub_det["lang"], dub_det["method"])
            gem_summary = ginfo.summarize(src_ext, dub_ext)
            thr = timing.speaker_threshold(op)
            drifts = timing.word_drifts(d_w, t, op, thr)
            pairs, missing, extra = align.align_words(src_w, d_w)

            starts = [abs(d_w[j]["start"] - src_w[i]["start"]) * 1000 for i, j in pairs]
            dub_speech = _speech_total(wav)
            values = {
                "translation_fidelity": align.lexical_fidelity(src_w, pairs),
                "word_count_diff": round(100.0 * abs(len(d_w) - len(src_w)) / max(len(src_w), 1), 1),
                "syllable_ratio": round(dub_speech / src_speech_s, 3) if src_speech_s else None,
                "timing_drift_ms": round(float(np.median(starts)), 1) if starts else None,
                "onset_drift_ms": timing.median_abs([d["onset_ms"] for d in drifts]),
                "tail_drift_ms": timing.median_abs([d["tail_ms"] for d in drifts]),
                "open_ratio_pct": timing.open_ratio_during_speech(t, op, thr, speech),
                "false_movement_pct": timing.false_movement(t, op, thr, speech),
                "word_pass_rate_pct": timing.word_pass_rate(drifts),
                "missing_sentences": align.missing_sentences(src_sent, src_w, pairs) if src_sent else None,
                "hallucinations": None,
                "emotion_match_pct": None,
                "tone_match_pct": None,
                "audio_quality": None,
            }
            extras = {"duration_s": info["duration_s"],
                      "duration_diff_pct": timing.duration_diff_pct(info["duration_s"], src_info["duration_s"]),
                      "wpm_ratio": timing.speech_rate_ratio(len(d_w), info["duration_s"], len(src_w),
                                                           src_info["duration_s"]),
                      "av_offset_ms": timing.av_offset_ms(t, op, speech),
                      "missing_words": missing[:50], "extra_words": extra[:50],
                      "translation_method": "lexical-proxy" if mode == "precomputed" else "lexical-proxy (COMET not wired yet)",
                      "speaker_threshold": None if np.isnan(thr) else round(float(thr), 4)}

            if use_judge and os.environ.get("GEMINI_API_KEY"):
                judged = _judge_outputs(run_dir, path, oid, results["segments"], d_w, src_w, values, extras)
                values.update(judged["values"])
                extras["judge_notes"] = judged["notes"]

            if values.get("emotion_match_pct") is None and gem_summary.get("emotion_match_pct") is not None:
                values["emotion_match_pct"] = gem_summary["emotion_match_pct"]
                extras["emotion_source"] = "Gemini labels, compared by code"
            results["info"][oid] = {
                "language": {"declared": man.get("target_lang"), "resolved": dub_lang,
                             "detection": dub_det, "notes": dub_lang_notes},
                "gemini": _gemini_brief(dub_ext),
                "summary": gem_summary,
            }
            _write(run_dir / "measure" / f"{oid}.json", {"drifts": drifts, "speech": speech, "lips_t_len": int(t.size)})
            values_by_out[oid] = values
            sc = scoring.score_output(values, th)
            log.info("%s done: overall=%s coverage=%s%%", oid, sc.get("overall"), sc.get("coverage_pct"))
            results["outputs"][oid] = {"output_id": oid, "vendor": o["vendor"], "file": o["file"],
                                       "values": values, "extras": extras, "score": sc}

        ranks = scoring.rank_outputs({k: v["score"]["overall"] for k, v in results["outputs"].items()})
        for k, r in ranks.items():
            results["outputs"][k]["overall_rank"] = r
        log.info("ranking and pairwise comparison")
        cmp = pairwise.compare_all(values_by_out, th)
        results["pairs"] = cmp["pairs"]
        results["win_matrix"] = cmp["win_matrix"]
        _write(run_dir / "results.json", results)
        _status(run_dir, "done")
        log.info("run finished")
        return results
    except Exception as e:  # keep the error visible in the status file
        log.exception("run failed")
        _status(run_dir, "failed", error=str(e))
        raise


def _speech_total(wav: Path) -> float:
    samples, sr = media.read_wav_mono(wav)
    return round(sum(e - s for s, e in vad.speech_intervals(samples, sr)), 3)


def _gemini_brief(ext: dict) -> dict:
    """Status of one Gemini extraction, without the full segment list (that is in the cache file)."""
    return {"status": ext.get("status"), "reason": ext.get("reason", ""), "model": ext.get("model"),
            "language": ext.get("language"), "segments": len(ext.get("segments", []) or []),
            "dropped_segments": ext.get("dropped_segments", 0)}


def _judge_outputs(run_dir, path, oid, segments, d_w, src_w, values, extras):
    notes, em, tm = [], [], []
    for seg in segments:
        clip = media.trim_clip(path, seg["start"], seg["end"], run_dir / "judge" / f"{seg['seg']:02d}__{oid}.mp4")
        dub_text = " ".join(w["word"] for w in d_w if seg["start"] <= w["start"] < seg["end"])
        measured = {"fidelity": values.get("translation_fidelity"), "onset_ms": values.get("onset_drift_ms")}
        prompt = judge_gemini.build_prompt(seg["text"], dub_text, measured)
        res = judge_gemini.judge_clip(clip, prompt) or {}
        notes.append({"seg": seg["seg"], **res})
        if res.get("status") == "ok":
            em.append(bool(res.get("emotion_match")))
            tm.append(bool(res.get("tone_match")))
    out = {"values": {}, "notes": notes}
    if em:
        out["values"]["emotion_match_pct"] = round(100.0 * sum(em) / len(em), 1)
    if tm:
        out["values"]["tone_match_pct"] = round(100.0 * sum(tm) / len(tm), 1)
    return out

core/align.py — Lexical Word Alignment & Matching

Fast Dynamic Time Warping (DTW) and edit distance matching between source and dubbed words, detecting missing and hallucinated words.
"""Word and sentence alignment between source and dub transcripts."""
import difflib
import unicodedata


def norm(word: str) -> str:
    """Lowercase, punctuation and symbols removed. Combining marks are kept, so Hindi words stay whole."""
    return "".join(ch for ch in word if not unicodedata.category(ch).startswith(("P", "S"))).lower()


def align_words(src_words: list[dict], dub_words: list[dict]):
    """Return matched pairs (src_idx, dub_idx) and lists of missing/extra source words."""
    a = [norm(w["word"]) for w in src_words]
    b = [norm(w["word"]) for w in dub_words]
    sm = difflib.SequenceMatcher(None, a, b, autojunk=False)
    pairs, matched_src = [], set()
    for block in sm.get_matching_blocks():
        for k in range(block.size):
            pairs.append((block.a + k, block.b + k))
            matched_src.add(block.a + k)
    missing = [src_words[i]["word"] for i in range(len(a)) if i not in matched_src]
    matched_dub = {j for _, j in pairs}
    extra = [dub_words[j]["word"] for j in range(len(b)) if j not in matched_dub]
    return pairs, missing, extra


def lexical_fidelity(src_words, pairs) -> float | None:
    if not src_words:
        return None
    return round(100.0 * len(pairs) / len(src_words), 1)


def missing_sentences(sentences: list[dict], src_words, pairs) -> int:
    """A source sentence is missing when less than half of its words are matched."""
    matched = {i for i, _ in pairs}
    missing = 0
    for s in sentences:
        idx = [i for i, w in enumerate(src_words) if s["start"] <= w["start"] < s["end"] + 1e-6]
        if idx and sum(1 for i in idx if i in matched) / len(idx) < 0.5:
            missing += 1
    return missing

core/timing.py — Drift & Synchronization Mathematics

Calculates onset drift (ms), tail drift (ms), false movement percentage during silence, open ratio, and audio-visual offsets.
"""Timing metrics. Pure numpy so every function can be unit-tested without models.

Units: time in seconds in, milliseconds out where the name says _ms.
"""
import numpy as np

from core.vad import in_any

ONSET_WINDOW = (-0.30, 0.50)   # search window around word start
TAIL_WINDOW = (-0.20, 0.50)    # search from slightly before word end (early close = negative tail)


def speaker_threshold(opening: np.ndarray) -> float:
    valid = opening[~np.isnan(opening)]
    if valid.size == 0:
        return float("nan")
    return 0.20 * float(np.percentile(valid, 95))


def word_drifts(words: list[dict], t: np.ndarray, opening: np.ndarray, thr: float):
    """Per word: onset_ms, tail_ms (None when no mouth event found in the window)."""
    out = []
    for w in words:
        s, e = float(w["start"]), float(w["end"])
        onset = None
        m = (t >= s + ONSET_WINDOW[0]) & (t <= s + ONSET_WINDOW[1]) & (opening > thr)
        if m.any():
            onset = (t[np.argmax(m)] - s) * 1000.0
        tail = None
        m2 = (t >= e + TAIL_WINDOW[0]) & (t <= e + TAIL_WINDOW[1]) & (opening < thr)
        if m2.any():
            tail = (t[np.argmax(m2)] - e) * 1000.0
        out.append({"word": w.get("word", ""), "start": s, "end": e,
                    "onset_ms": None if onset is None else round(onset, 1),
                    "tail_ms": None if tail is None else round(tail, 1)})
    return out


def word_pass_rate(drifts: list[dict], good_ms: float = 80.0):
    scored = [d for d in drifts if d["onset_ms"] is not None and d["tail_ms"] is not None]
    if not drifts:
        return None
    ok = sum(1 for d in scored if abs(d["onset_ms"]) < good_ms and abs(d["tail_ms"]) < good_ms)
    return round(100.0 * ok / len(drifts), 1)   # unmatched words count as fails


def median_abs(values) -> float | None:
    vals = [abs(v) for v in values if v is not None]
    return round(float(np.median(vals)), 1) if vals else None


def open_ratio_during_speech(t, opening, thr, speech) -> float | None:
    mask = np.array([in_any(x, speech) for x in t])
    if not mask.any():
        return None
    return round(100.0 * float(np.mean(opening[mask] > thr)), 1)


def false_movement(t, opening, thr, speech) -> float | None:
    mask = np.array([not in_any(x, speech) for x in t])
    if not mask.any():
        return None
    return round(100.0 * float(np.mean(opening[mask] > thr)), 1)


def av_offset_ms(t_lips, opening, speech, max_lag_ms: float = 500.0) -> float | None:
    """Lag (ms) that best aligns the lip-open signal with the speech signal.
    Positive = mouth moves after the audio."""
    if t_lips.size < 4 or not speech:
        return None
    grid = np.arange(0, t_lips[-1], 0.01)
    if grid.size < 4:
        return None
    speech_sig = np.array([1.0 if in_any(x, speech) else 0.0 for x in grid])
    lip = np.interp(grid, t_lips, np.nan_to_num(opening, nan=0.0))
    a = speech_sig - speech_sig.mean()
    b = lip - lip.mean()
    lags = np.arange(-int(max_lag_ms / 10), int(max_lag_ms / 10) + 1)
    best, best_lag = -np.inf, 0
    for k in lags:
        if k >= 0:
            x, y = a[: a.size - k], b[k:]
        else:
            x, y = a[-k:], b[: b.size + k]
        if x.size == 0:
            continue
        c = float(np.dot(x, y) / (np.linalg.norm(x) * np.linalg.norm(y) + 1e-9))
        if c > best:
            best, best_lag = c, k
    return float(best_lag * 10)


def speech_rate_ratio(dub_words: int, dub_dur: float, src_words: int, src_dur: float):
    if min(dub_dur, src_dur, src_words) <= 0:
        return None
    return round((dub_words / dub_dur) / (src_words / src_dur), 3)


def duration_diff_pct(dub_dur: float, src_dur: float) -> float | None:
    if src_dur <= 0:
        return None
    return round(100.0 * abs(dub_dur - src_dur) / src_dur, 2)

core/scoring.py — Weighted Multi-Metric Scoring

Applies YAML thresholds and metric group weights to compute overall scores (0-100) and coverage percentages without vendor bias.
"""Turn measured values into Good / Check / Fix labels and an overall score.

Missing values (None) are never treated as a pass. They are excluded and
reported through coverage.
"""
from pathlib import Path

import yaml

ROOT = Path(__file__).resolve().parent.parent


def load_thresholds(path: Path | None = None) -> dict:
    p = path or ROOT / "config" / "thresholds.yaml"
    return yaml.safe_load(p.read_text(encoding="utf-8"))


def label(key: str, value, th: dict) -> str | None:
    if value is None:
        return None
    m = th["metrics"][key]
    d = m["direction"]
    if d == "min":
        return "Good" if value >= m["good"] else ("Check" if value >= m["check"] else "Fix")
    if d == "max":
        return "Good" if value <= m["good"] else ("Check" if value <= m["check"] else "Fix")
    if d == "max_abs":
        v = abs(value)
        return "Good" if v <= m["good"] else ("Check" if v <= m["check"] else "Fix")
    if d == "band":
        if m["low"] <= value <= m["high"]:
            return "Good"
        if m["check_low"] <= value <= m["check_high"]:
            return "Check"
        return "Fix"
    raise ValueError(f"unknown direction for {key}: {d}")


POINTS = {"Good": 1.0, "Check": 0.5, "Fix": 0.0}


def score_output(values: dict, th: dict) -> dict:
    """values: metric_key -> measured value (or None). Returns labels, group scores, overall."""
    labels = {k: label(k, values.get(k), th) for k in th["metrics"]}
    groups: dict[str, list[float]] = {}
    for k, lab in labels.items():
        if lab is None:
            continue
        g = th["metrics"][k]["group"]
        groups.setdefault(g, []).append(POINTS[lab])
    group_scores = {g: round(100.0 * sum(v) / len(v), 1) for g, v in groups.items()}
    weights = th["group_weights"]
    used = {g: weights[g] for g in group_scores if g in weights}
    total_w = sum(used.values())
    overall = round(sum(group_scores[g] * w for g, w in used.items()) / total_w, 1) if total_w else None
    measured = sum(1 for lab in labels.values() if lab is not None)
    coverage = round(100.0 * measured / len(labels), 1)
    fixes = [k for k, lab in labels.items() if lab == "Fix"]
    gate = th["release_gate"]
    critical_fail = any(k in ("missing_sentences", "hallucinations") and labels[k] == "Fix" for k in labels)
    approved = (overall is not None and overall >= gate["min_overall"]
                and coverage >= 100 - gate["max_missing_coverage_pct"]
                and not critical_fail and not fixes)
    return {"labels": labels, "group_scores": group_scores, "overall": overall,
            "coverage_pct": coverage, "fix_metrics": fixes,
            "release": "Approved" if approved else ("Review" if overall is not None else "Not measured")}


def rank_outputs(scores: dict[str, float | None]) -> dict[str, str]:
    """Competition ranking with ties (1, 1, 3). None ranks last with no rank."""
    ranked = sorted([(v, k) for k, v in scores.items() if v is not None], key=lambda x: -x[0])
    out, prev, pos = {}, None, 0
    for i, (v, k) in enumerate(ranked, start=1):
        if v != prev:
            pos = i
            prev = v
        tied = sum(1 for vv, _ in ranked if vv == v)
        out[k] = f"{pos} (tie)" if tied > 1 else str(pos)
    for k, v in scores.items():
        if v is None:
            out[k] = "—"
    return out

core/pairwise.py — Head-to-Head Pairwise Comparison Matrix

Compares every dubbed output against every other output across all metrics to compute wins, losses, ties, and tournament rankings.
"""Every dub against every other dub. Winner per metric uses the tie tolerances."""
from itertools import combinations


def better(key: str, a, b, th: dict) -> str:
    """Return 'a', 'b' or 'tie' for one metric."""
    if a is None or b is None:
        return "n/a"
    m = th["metrics"][key]
    tol = th["tie_tolerance"].get(key, 0)
    d = m["direction"]
    if d == "band":
        mid = (m["low"] + m["high"]) / 2
        a2, b2 = abs(a - mid), abs(b - mid)
        lower_better = True
        a, b = a2, b2
    elif d in ("max", "max_abs"):
        lower_better = True
        if d == "max_abs":
            a, b = abs(a), abs(b)
    else:
        lower_better = False
    if abs(a - b) <= tol:
        return "tie"
    if lower_better:
        return "a" if a < b else "b"
    return "a" if a > b else "b"


def compare_all(outputs: dict[str, dict], th: dict) -> dict:
    """outputs: output_id -> metric values. Returns pair results and a win matrix."""
    ids = list(outputs)
    pairs = {}
    wins = {i: {j: 0 for j in ids} for i in ids}
    for a, b in combinations(ids, 2):
        per_metric = {}
        score_a = score_b = 0
        for k in th["metrics"]:
            r = better(k, outputs[a].get(k), outputs[b].get(k), th)
            if r == "a":
                score_a += 1
                wins[a][b] += 1
                per_metric[k] = a
            elif r == "b":
                score_b += 1
                wins[b][a] += 1
                per_metric[k] = b
            else:
                per_metric[k] = r
        pairs[f"{a} vs {b}"] = {"a": a, "b": b, "wins_a": score_a, "wins_b": score_b,
                                "per_metric": per_metric}
    return {"pairs": pairs, "win_matrix": wins}

core/judge_gemini.py — Gemini Per-Segment Emotion Judge

Calls Gemini with video segment payloads to evaluate emotion match, tone preservation, and semantic integrity.

Timing numbers are never produced here. Gemini returns labels and transcripts; code measures.
Skipped cleanly when GEMINI_API_KEY or google-genai is missing.

import json

from pathlib import Path

SCHEMA_HINT = """Return ONLY JSON with this shape:
{"emotion_src": "", "emotion_dub": "", "emotion_match": true,
 "tone_src": "", "tone_dub": "", "tone_match": true,
 "meaning_preserved": true,
 "issues": [{"time_s": 0.0, "type": "", "note": ""}],
 "summary": ""}"""

        + SCHEMA_HINT
    )


def gemini_available() -> tuple[bool, str]:
    if not os.environ.get("GEMINI_API_KEY"):
        return False, "GEMINI_API_KEY not set"
    try:
        import google.genai  # noqa: F401
    except ImportError:
        return False, "google-genai not installed"
    return True, "ok"


def call_gemini_json(media_path: Path, prompt: str) -> dict:
    """Upload a video or audio file, ask for JSON, return the parsed object plus status."""
    ok, reason = gemini_available()
    if not ok:
        return {"status": "skipped", "reason": reason}
    from google import genai
    client = genai.Client(api_key=os.environ["GEMINI_API_KEY"])
    f = client.files.upload(file=str(media_path))
    for _ in range(60):
        if client.files.get(name=f.name).state.name == "ACTIVE":
            break
        time.sleep(2)
    model = os.environ.get("GEMINI_MODEL", "gemini-3.8-flash")
    resp = client.models.generate_content(
        model=model, contents=[f, prompt],
        config={"response_mime_type": "application/json"})
    try:
        out = json.loads(resp.text)
    except (json.JSONDecodeError, TypeError):
        return {"status": "parse_error", "raw": str(resp.text)[:500], "model": model}
    if not isinstance(out, dict):
        return {"status": "parse_error", "raw": str(resp.text)[:500], "model": model}
    out["status"] = "ok"
    out["model"] = model
    return out


def judge_clip(clip: Path, prompt: str) -> dict:
    return call_gemini_json(clip, prompt)

core/vad.py — Voice Activity Detection (Energy VAD)

Computes audio RMS energy envelope to detect voiced speech intervals and inter-word pauses without external AI models.
"""Energy-based voice activity detection (no model needed).

Frames of 20 ms. A frame is speech when its RMS is above an adaptive floor.
Returns merged speech intervals in seconds.
"""
import numpy as np


def speech_intervals(samples: np.ndarray, sr: int, frame_ms: int = 20,
                     rel_db: float = -30.0, min_len_s: float = 0.08,
                     merge_gap_s: float = 0.15) -> list[tuple[float, float]]:
    if samples.size == 0:
        return []
    n = int(sr * frame_ms / 1000)
    n_frames = samples.size // n
    if n_frames == 0:
        return []
    frames = samples[: n_frames * n].reshape(n_frames, n)
    rms = np.sqrt((frames ** 2).mean(axis=1)) + 1e-9
    ref = np.percentile(rms, 95)
    thr = ref * (10 ** (rel_db / 20.0))
    active = rms > thr
    intervals, start = [], None
    for i, a in enumerate(active):
        if a and start is None:
            start = i
        if not a and start is not None:
            intervals.append([start, i]); start = None
    if start is not None:
        intervals.append([start, n_frames])
    dt = frame_ms / 1000.0
    secs = [(s * dt, e * dt) for s, e in intervals]
    # merge close intervals
    merged: list[list[float]] = []
    for s, e in secs:
        if merged and s - merged[-1][1] <= merge_gap_s:
            merged[-1][1] = e
        else:
            merged.append([s, e])
    return [(round(s, 3), round(e, 3)) for s, e in merged if e - s >= min_len_s]


def in_any(t: float, intervals) -> bool:
    return any(s <= t < e for s, e in intervals)

core/media.py — FFmpeg Audio & Video Utilities

Probes video metadata, extracts 16kHz mono WAV streams, and reads raw PCM buffers for VAD and signal processing.
"""FFmpeg / ffprobe helpers. All media work goes through here."""
import json
import subprocess
from pathlib import Path


def _run(cmd: list[str]) -> subprocess.CompletedProcess:
    res = subprocess.run(cmd, capture_output=True, text=True)
    if res.returncode != 0:
        raise RuntimeError(f"command failed: {' '.join(cmd)}\n{res.stderr[-800:]}")
    return res


def probe(path: Path) -> dict:
    res = _run(["ffprobe", "-v", "error", "-print_format", "json",
                "-show_format", "-show_streams", str(path)])
    info = json.loads(res.stdout)
    streams = info.get("streams", [])
    v = next((s for s in streams if s.get("codec_type") == "video"), None)
    a = next((s for s in streams if s.get("codec_type") == "audio"), None)
    return {
        "duration_s": round(float(info["format"].get("duration", 0.0)), 3),
        "has_video": v is not None,
        "has_audio": a is not None,
        "width": v.get("width") if v else None,
        "height": v.get("height") if v else None,
        "fps": v.get("r_frame_rate") if v else None,
    }


def extract_audio_wav(src: Path, dst: Path, sr: int = 16000) -> Path:
    dst.parent.mkdir(parents=True, exist_ok=True)
    _run(["ffmpeg", "-y", "-i", str(src), "-vn", "-ac", "1", "-ar", str(sr),
          "-acodec", "pcm_s16le", str(dst)])
    return dst


def read_wav_mono(path: Path):
    """Return (samples float32 in [-1,1], sample_rate) using the stdlib wave module."""
    import wave
    import numpy as np
    with wave.open(str(path), "rb") as w:
        sr = w.getframerate()
        raw = w.readframes(w.getnframes())
        width = w.getsampwidth()
    dtype = {1: np.uint8, 2: np.int16, 4: np.int32}[width]
    arr = np.frombuffer(raw, dtype=dtype).astype(np.float32)
    if width == 1:
        arr = (arr - 128.0) / 128.0
    else:
        arr = arr / float(np.iinfo(dtype).max)
    return arr, sr


def trim_clip(src: Path, start_s: float, end_s: float, dst: Path) -> Path:
    dst.parent.mkdir(parents=True, exist_ok=True)
    _run(["ffmpeg", "-y", "-ss", f"{start_s:.3f}", "-to", f"{end_s:.3f}", "-i", str(src),
          "-c:v", "libx264", "-preset", "veryfast", "-crf", "30", "-c:a", "aac", "-b:a", "64k",
          "-movflags", "+faststart", str(dst)])
    return dst


def make_web_copy(src: Path, dst: Path, height: int = 360) -> Path:
    dst.parent.mkdir(parents=True, exist_ok=True)
    _run(["ffmpeg", "-y", "-i", str(src), "-vf", f"scale=-2:{height}",
          "-c:v", "libx264", "-preset", "veryfast", "-crf", "32", "-c:a", "aac", "-b:a", "48k",
          "-ac", "1", "-movflags", "+faststart", str(dst)])
    return dst

core/lang.py — Language Detection & Resolution

Combines ASR declared language with unicode script frequency heuristics and Gemini confidence to resolve spoken languages.
"""Language detection and resolution.

Order of evidence for each video:
1. Whisper's detected language (full mode), or a "language" field in the words JSON (precomputed).
2. Gemini's language field from the extraction (when Gemini ran).
3. Unicode script of the transcript words. This identifies non-Latin scripts reliably
   (Devanagari -> hi, Tamil -> ta, ...). Latin script cannot separate en, es, fr, de, so it
   is reported as undetermined and the declared value is kept.

A declared language in the manifest wins unless it is "auto" or missing. A disagreement
with the detected language is recorded as a note, never silently overwritten.
"""
from collections import Counter

# Script name, code, Unicode block. Devanagari covers hi, mr, ne: script alone cannot separate them.
SCRIPTS = [
    ("Devanagari", "hi", 0x0900, 0x097F),
    ("Bengali", "bn", 0x0980, 0x09FF),
    ("Gurmukhi", "pa", 0x0A00, 0x0A7F),
    ("Gujarati", "gu", 0x0A80, 0x0AFF),
    ("Tamil", "ta", 0x0B80, 0x0BFF),
    ("Telugu", "te", 0x0C00, 0x0C7F),
    ("Kannada", "kn", 0x0C80, 0x0CFF),
    ("Malayalam", "ml", 0x0D00, 0x0D7F),
    ("Thai", "th", 0x0E00, 0x0E7F),
    ("Arabic", "ar", 0x0600, 0x06FF),
    ("Greek", "el", 0x0370, 0x03FF),
    ("Cyrillic", "ru", 0x0400, 0x04FF),
    ("Hangul", "ko", 0xAC00, 0xD7AF),
    ("Kana", "ja", 0x3040, 0x30FF),
    ("Han", "zh", 0x4E00, 0x9FFF),
]
LATIN_RANGES = [(0x0041, 0x005A), (0x0061, 0x007A), (0x00C0, 0x024F)]


def _is_latin(ch: str) -> bool:
    c = ord(ch)
    return any(lo <= c <= hi for lo, hi in LATIN_RANGES)


def detect_from_text(text: str) -> dict:
    counts: Counter = Counter()
    latin = 0
    for ch in text:
        if not ch.isalpha():
            continue
        c = ord(ch)
        hit = None
        for name, code, lo, hi in SCRIPTS:
            if lo <= c <= hi:
                hit = (name, code)
                break
        if hit:
            counts[hit] += 1
        elif _is_latin(ch):
            latin += 1
    total = sum(counts.values()) + latin
    if total == 0:
        return {"lang": None, "method": "script", "script": None, "confidence": 0.0,
                "note": "no letters to detect from"}
    if counts:
        (name, code), n = counts.most_common(1)[0]
        share = round(n / total, 3)
        if share >= 0.5:
            note = "Devanagari is shared by hi, mr and ne" if code == "hi" else ""
            return {"lang": code, "method": "script", "script": name, "confidence": share, "note": note}
    return {"lang": None, "method": "script", "script": "Latin", "confidence": round(latin / total, 3),
            "note": "Latin script: cannot tell English, Spanish, French or German from script alone"}


def resolve(declared: str | None, detected: str | None) -> tuple[str | None, list[str]]:
    """Return the language to use and any notes. Declared wins unless it is 'auto' or missing."""
    notes: list[str] = []
    decl = (declared or "").strip().lower()
    if decl and decl != "auto":
        if detected and detected != decl:
            notes.append(f"declared '{decl}' but detected '{detected}'; declared value used")
        return decl, notes
    if detected:
        return detected, notes
    notes.append("language undetermined; set it in the form if known")
    return None, notes


def detect_for_words(words: list[dict], language_field: str | None) -> dict:
    """Use an explicit language field when present, otherwise fall back to script detection."""
    if language_field:
        return {"lang": language_field.lower(), "method": "provided", "script": None,
                "confidence": 1.0, "note": ""}
    text = " ".join(w.get("word", "") for w in words)
    return detect_from_text(text)

core/manifest.py — Run Manifest Validator

Loads and validates run_id, source video specs, and unique output identifiers against schema constraints.
"""Read and validate a batch manifest."""
import json
import re
from pathlib import Path

NAME_RE = re.compile(r"^[a-z0-9_\-]+$")


class ManifestError(ValueError):
    pass


def load_manifest(path: Path) -> dict:
    data = json.loads(Path(path).read_text(encoding="utf-8"))
    validate_manifest(data)
    return data


def validate_manifest(data: dict) -> None:
    for key in ("run_id", "source", "target_lang", "outputs"):
        if key not in data:
            raise ManifestError(f"missing key: {key}")
    if not data["outputs"]:
        raise ManifestError("no outputs listed")
    ids = set()
    for o in data["outputs"]:
        for k in ("output_id", "vendor", "file"):
            if k not in o:
                raise ManifestError(f"output missing {k}: {o}")
        if not NAME_RE.match(o["output_id"]):
            raise ManifestError(f"bad output_id (use lowercase, digits, _ or -): {o['output_id']}")
        if o["output_id"] in ids:
            raise ManifestError(f"duplicate output_id: {o['output_id']}")
        ids.add(o["output_id"])

    thread_name = threading.current_thread().name
    handler = logging.FileHandler(run_dir / LOG_NAME, encoding="utf-8")
    handler.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(message)s"))
    handler.addFilter(_ThreadFilter(thread_name))
    logger = logging.getLogger("dubqa")
    logger.addHandler(handler)
    try:
        pipeline.run(run_dir, mode=mode, use_judge=use_judge, use_gemini=use_gemini)
        build_html.build(run_dir)
        logging.getLogger("dubqa").info("report built")
    except Exception:
        pass  # pipeline already wrote status=failed with the error, and the traceback is in run.log
    finally:
        logger.removeHandler(handler)
        handler.close()

@app.post("/api/runs", status_code=202)
async def create_run(manifest: UploadFile = File(...), files: list[UploadFile] = File(...),
                     mode: str = Form("precomputed"), use_judge: bool = Form(False),
                     use_gemini: bool = Form(True)):
    if mode not in ("precomputed", "full"):
        raise HTTPException(400, "mode must be precomputed or full")
    raw = await manifest.read()
    try:
        data = json.loads(raw)
        load_manifest_from_dict(data)
    except (ValueError, ManifestError) as e:
        raise HTTPException(400, f"manifest invalid: {e}")
    run_id = data["run_id"]
    if not RUN_ID_RE.match(run_id):
        raise HTTPException(400, "bad run_id in manifest")
    run_dir = RUNS / run_id
    if run_dir.exists():
        raise HTTPException(409, "run_id already exists")
    for sub in ("inputs", "work", "measure", "judge", "precomputed", "report"):
        (run_dir / sub).mkdir(parents=True, exist_ok=True)
    (run_dir / "manifest.json").write_bytes(raw)
    wanted = {data["source"]["file"], *(o["file"] for o in data["outputs"])}
    saved = []
    for f in files:
        name = Path(f.filename or "").name
        if not name or name != (f.filename or ""):
            raise HTTPException(400, f"bad file name: {f.filename}")
        if name.endswith(".json"):
            dest = run_dir / "precomputed" / name
        elif name in wanted:
            dest = run_dir / "inputs" / name
        else:
            raise HTTPException(400, f"unexpected file: {name}")
        dest.write_bytes(await f.read())
        saved.append(name)
    missing = [w for w in wanted if w not in saved]
    if missing:
        raise HTTPException(400, f"missing media files: {missing}")
    _write_meta(run_dir, "queued", mode=mode, created=datetime.now(timezone.utc).isoformat())
    (run_dir / LOG_NAME).write_text(
        f"{datetime.now(timezone.utc).isoformat()} INFO run {run_id} queued (mode={mode})\n", encoding="utf-8")
    t = threading.Thread(target=_job, args=(run_dir, mode, use_judge, use_gemini), name=f"run-{run_id}", daemon=True)
    t.start()
    return {"run_id": run_id, "status": "queued", "status_url": f"/api/runs/{run_id}"}


def load_manifest_from_dict(data: dict) -> None:
    from core.manifest import validate_manifest

def _write_meta(run_dir: Path, status: str, **extra) -> None:
    p = run_dir / "run_meta.json"
    meta = json.loads(p.read_text()) if p.exists() else {}
    meta.update({"status": status}, **extra)
    p.write_text(json.dumps(meta, indent=2))

@app.get("/api/runs/{run_id}/log")
def run_log(run_id: str, offset: int = Query(0, ge=0)):
    p = _run_dir(run_id) / LOG_NAME
    if not p.exists():
        return {"text": "", "offset": 0}
    size = p.stat().st_size
    start = min(offset, size)
    with p.open("rb") as f:
        f.seek(start)
        chunk = f.read(MAX_LOG_CHUNK)
    return {"text": chunk.decode("utf-8", errors="replace"), "offset": start + len(chunk)}
